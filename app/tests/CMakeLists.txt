find_package(Qt6 REQUIRED COMPONENTS Test)

function(vial_armed_test name)
    qt_add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE vialarmed_core Qt6::Test)
    target_compile_definitions(${name} PRIVATE VIAL_ARMED_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/data")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()

vial_armed_test(tst_appinfo)
vial_armed_test(tst_link)
vial_armed_test(tst_session)
vial_armed_test(tst_vial)
vial_armed_test(tst_armed)
vial_armed_test(tst_keycodes)
vial_armed_test(tst_definition)
vial_armed_test(tst_macro)
vial_armed_test(tst_hud)
vial_armed_test(tst_lighting)
vial_armed_test(tst_screenmedia)
